cmake_minimum_required(VERSION 3.16)
project(implicit_heat LANGUAGES CXX)
find_package(MPI REQUIRED)
add_executable(implicit_heat src/main.cpp)
target_compile_features(implicit_heat PRIVATE cxx_std_17)
target_link_libraries(implicit_heat PRIVATE MPI::MPI_CXX)
target_compile_options(implicit_heat PRIVATE -Wall -Wextra)
add_executable(test_stencil tests/test_stencil.cpp)
target_compile_features(test_stencil PRIVATE cxx_std_17)
target_link_libraries(test_stencil PRIVATE MPI::MPI_CXX)
target_compile_options(test_stencil PRIVATE -Wall -Wextra)
add_executable(implicit_heat_diagnostics src/main.cpp)
target_compile_features(implicit_heat_diagnostics PRIVATE cxx_std_17)
target_compile_definitions(implicit_heat_diagnostics PRIVATE HEAT_DIAGNOSTICS)
target_link_libraries(implicit_heat_diagnostics PRIVATE MPI::MPI_CXX)
target_compile_options(implicit_heat_diagnostics PRIVATE -Wall -Wextra)
